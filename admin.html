function publish(){
  sync();
  const blob = new Blob([generateFile()], { type: 'text/javascript' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'moon-config.js';
  a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 4000);
}

function importConfig(ev){
  const file = ev.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(){
    try{
      const w = {};
      new Function('window', reader.result)(w);
      if (w.CFG){
        cfg = Object.assign(clone(DEFAULTS), w.CFG);
        if (w.CFG.CONTACT) cfg.CONTACT = Object.assign(clone(DEFAULTS.CONTACT), w.CFG.CONTACT);
        if (w.CFG.ADS)     cfg.ADS     = Object.assign(clone(DEFAULTS.ADS),     w.CFG.ADS);
        fillForm();
        alert('✓ Imported. Review the fields, then click Publish.');
      } else alert('Could not find window.CFG in that file.');
    }catch(e){ alert('Import failed: ' + e.message); }
  };
  reader.readAsText(file);
  ev.target.value = '';
}
